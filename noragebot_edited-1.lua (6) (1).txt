local function _mkCosmetic(nm, ctype, opts)
    local _base = _cosLib.Cosmetics[nm]
    if not _base then return nil end
    local _d = {}
    for k, v in pairs(_base) do _d[k] = v end
    _d.Name = nm
    _d.Type = _d.Type or ctype
    _d.Seed = _d.Seed or math.random(1, 1000000)
    if _enumLib then
        local _s, _eid = pcall(_enumLib.ToEnum, _enumLib, nm)
        if _s and _eid then
            _d.Enum = _eid
            _d.ObjectID = _d.ObjectID or _eid
        end
    end
    if opts then
        if opts.inverted ~= nil then _d.Inverted = opts.inverted end
        if opts.favoritesOnly ~= nil then _d.OnlyUseFavorites = opts.favoritesOnly end
    end
    return _d
end

local _cfgFile = "rivals_unlocker_config.json"
local _saveLock = false

local function _stripForSave()
    local _out = {}
    for wn, cos in pairs(_eq) do
        _out[wn] = {}
        for ct, cd in pairs(cos) do
            if cd and cd.Name then
                _out[wn][ct] = {
                    Name = cd.Name,
                    Inverted = cd.Inverted,
                    OnlyUseFavorites = cd.OnlyUseFavorites
                }
            end
        end
    end
    return { equipped = _out, favorites = _favs }
end

local function _loadCfg()
    if not isfile or not readfile then return end
    local _ok1, _ex = pcall(isfile, _cfgFile)
    if not _ok1 or not _ex then return end
    local _ok2, _raw = pcall(readfile, _cfgFile)
    if not _ok2 or not _raw or _raw == "" then return end
    local _ok3, _dec = pcall(_http.JSONDecode, _http, _raw)
    if not _ok3 or not _dec then return end
    if _dec.favorites then
        _favs = _dec.favorites
    end
    if _dec.equipped then
        _eq = {}
        local _cnt = 0
        for wn, cos in pairs(_dec.equipped) do
            _eq[wn] = {}
            for ct, sd in pairs(cos) do
                if sd and sd.Name then
                    if _cosLib.Cosmetics[sd.Name] then
                        local _cloned = _mkCosmetic(sd.Name, ct, {
                            inverted = sd.Inverted,
                            favoritesOnly = sd.OnlyUseFavorites
                        })
                        if _cloned then
                            _eq[wn][ct] = _cloned
                            _cnt += 1
                        end
                    end
                end
            end
            if not next(_eq[wn]) then _eq[wn] = nil end
        end
    end
end

local function _saveCfg()
    if not writefile or _saveLock then return end
    _saveLock = true
    task.spawn(function()
        task.wait(1)
        local _payload = _stripForSave()
        local _ok, _enc = pcall(_http.JSONEncode, _http, _payload)
        if _ok then
            pcall(writefile, _cfgFile, _enc)
        end
        _saveLock = false
    end)
end

            for k, v in pairs(_val) do
                local c = _cosLib.Cosmetics[k]
                if c and _isCosType(c) then _prx[k] = v end

        end
        return setmetatable(_prx, {
            __index = function(t, k)
                local c = _cosLib.Cosmetics[k]

                return nil
            end

                if c and _isCosType(c) then
                    _res[wep][nm] = isFav
                end
            end
        end
        return _res
    end
    return _val

local AntiKatanaModule = (function()
    local Players = game:GetService("Players")
    local Workspace = game:GetService("Workspace")

    local antikatana = false
    local katanausers = {}

    local function detectkatana()
        local lp = Players.LocalPlayer
        if not lp:FindFirstChild("PlayerScripts") then
            lp.PlayerScriptsAdded:Wait()
        end
        task.spawn(function()
            local katana, attempts = nil, 0
            while attempts < 10 do
                pcall(function()
                    local m = lp.PlayerScripts.Modules.Items:FindFirstChild("Katana", true)
                    if m then katana = require(m) end
                end)
                if not katana then
                    for _, m in pairs(lp.PlayerScripts:GetDescendants()) do
                        if m.Name == "Katana" and m:IsA("ModuleScript") then
                            local ok, res = pcall(require, m)
                            if ok then katana = res; break end
                        end
                    end
                end
                if katana and type(katana) == "table" and katana.StartAiming then break end
                attempts = attempts + 1
                task.wait(1)
            end
            if katana and type(katana) == "table" and katana.StartAiming then
                local old = katana.StartAiming
                katana.StartAiming = function(self, force)
                    local fighter = self.ClientFighter
                    local player = fighter and fighter.Player
                    if player then
                        katanausers[player] = true
                        local dur = self.Info.DeflectDuration or 0.6
                        task.delay(dur, function() katanausers[player] = nil end)
                    end
                    return old(self, force)
                end
            end
        end)
    end

    local function katanadeflect(player)
        return antikatana and (katanausers[player] == true)
    end

    local function setAntiKatana(enabled)
        antikatana = enabled
    end

    detectkatana()

    return {
        setAntiKatana = setAntiKatana,
        katanadeflect = katanadeflect,
        isEnabled = function() return antikatana end,
    }
end)()

-- ─────────────────────────────────────────────────────────────────
-- AlwaysBackstab module
-- Hooks KnifeItem methods so the server always sees yaw+180 / pitch=0
-- while the local knife is equipped, producing a server-side backstab.
-- Exposes getgenv().__AlwaysBackstab for console access.
-- ─────────────────────────────────────────────────────────────────
local AlwaysBackstabModule = (function()
    local _absPlayers  = game:GetService("Players")
    local _absRS       = game:GetService("RunService")
    local _absRepStor  = game:GetService("ReplicatedStorage")
    local _absLP       = _absPlayers.LocalPlayer

    -- ── lazy-load helpers so this module survives early injection ──
    local function _absRequireSafe(path)
        local ok, res = pcall(require, path)
        return ok and res or nil
    end

    local function _absGetKnifeItem()
        local ps = _absLP:FindFirstChild("PlayerScripts")
        if not ps then return nil end
        local mods = ps:FindFirstChild("Modules")
        if not mods then return nil end
        local items = mods:FindFirstChild("Items")
        if not items then return nil end
        local m = items:FindFirstChild("Knife")
        if not m then return nil end
        return _absRequireSafe(m)
    end

    local function _absFindMethod(tbl, candidates)
        for _, name in ipairs(candidates) do
            if type(tbl[name]) == "function" then return name end
        end
        return nil
    end

    -- ── utility / camera refs, same approach as the AA module ──
    local function _absGetUtility()
        local ok, u = pcall(require, _absRepStor:WaitForChild("Modules"):WaitForChild("Utility", 5))
        return ok and u or nil
    end
    local function _absGetCameraController()
        local ps = _absLP:FindFirstChild("PlayerScripts")
        if not ps then return nil end
        local ctrl = ps:FindFirstChild("Controllers")
        if not ctrl then return nil end
        local m = ctrl:FindFirstChild("CameraController")
        return m and _absRequireSafe(m) or nil
    end
    local function _absGetUpdateRemote()
        local rem = _absRepStor:FindFirstChild("Remotes")
        local rep = rem and rem:FindFirstChild("Replication")
        local fig = rep and rep:FindFirstChild("Fighter")
        return fig and fig:FindFirstChild("UpdateCameraRotation") or nil
    end

    -- ── module state ──
    local State = {
        Enabled       = false,
        HeartbeatConn = nil,
        Hooks         = {},
        KnifeItem     = nil,
    }

    local function _absIsLocalKnife(item)
        local fighter = item and item.ClientFighter
        if not fighter then return false end
        if fighter.IsLocalPlayer == true then return true end
        return fighter.Player == _absLP
    end

    local function _absStopLock()
        State.Enabled = false
        if State.HeartbeatConn then
            State.HeartbeatConn:Disconnect()
            State.HeartbeatConn = nil
        end
    end

    local function _absStartLock()
        if State.HeartbeatConn then
            State.HeartbeatConn:Disconnect()
            State.HeartbeatConn = nil
        end
        State.Enabled = true

        State.HeartbeatConn = _absRS.Heartbeat:Connect(function()
            if not State.Enabled then
                State.HeartbeatConn:Disconnect()
                State.HeartbeatConn = nil
                return
            end
            local util = _absGetUtility()
            local camCtrl = _absGetCameraController()
            local remote  = _absGetUpdateRemote()
            if not (util and camCtrl and remote) then return end

            local currentYaw = (camCtrl.Rotation and camCtrl.Rotation.Y) or 0
            local encoded = pcall(function()
                return util:EncodeCameraRotation(
                    Vector2.new(math.rad(0), currentYaw + math.rad(180))
                )
            end)
            -- fire raw if EncodeCameraRotation worked
            pcall(function()
                local enc = util:EncodeCameraRotation(
                    Vector2.new(math.rad(0), currentYaw + math.rad(180))
                )
                remote:FireServer(enc, nil)
            end)
        end)
    end

    local function _absEnsureLock()
        if not State.Enabled then _absStartLock() end
    end

    local function _absHookMethod(ki, key, onCall)
        if not key or not ki then return end
        local original = ki[key]
        State.Hooks[key] = original
        ki[key] = function(self, ...)
            if _absIsLocalKnife(self) then onCall() end
            return original(self, ...)
        end
    end

    local function _absInstallHooks(ki)
        if not ki then return end
        State.KnifeItem = ki
        State.Hooks = {}

        local EQUIP_KEY    = _absFindMethod(ki, {"Equip",    "equip",    "OnEquip",    "onEquip"})
        local UNEQUIP_KEY  = _absFindMethod(ki, {"Unequip",  "unequip",  "OnUnequip",  "onUnequip"})
        local ACTIVATE_KEY = _absFindMethod(ki, {"Activate", "activate", "OnActivate", "onActivate", "Use", "use"})
        local AIMING_KEY   = _absFindMethod(ki, {"StartAiming", "startAiming", "BeginAim", "beginAim"})

        _absHookMethod(ki, EQUIP_KEY,    _absStartLock)
        _absHookMethod(ki, UNEQUIP_KEY,  _absStopLock)
        _absHookMethod(ki, ACTIVATE_KEY, _absEnsureLock)
        _absHookMethod(ki, AIMING_KEY,   _absEnsureLock)
    end

    -- initial hook attempt
    task.spawn(function()
        local ki = _absGetKnifeItem()
        if ki then
            _absInstallHooks(ki)
        end
    end)

    -- ── public API ──
    State.Toggle = function()
        if State.Enabled then
            _absStopLock()
        else
            _absStartLock()
        end
    end

    State.Restore = function()
        _absStopLock()
        if State.KnifeItem then
            for key, original in pairs(State.Hooks) do
                State.KnifeItem[key] = original
            end
        end
        State.Hooks = {}
        State.KnifeItem = nil
    end

    -- re-hook after respawn (called from CharacterAdded)
    State.Rehook = function()
        State.Restore()
        task.spawn(function()
            task.wait(1)
            local ki = _absGetKnifeItem()
            if ki then _absInstallHooks(ki) end
            if S.alwaysBackstabActive then _absStartLock() end
        end)
    end

    State.SetEnabled = function(on)
        S.alwaysBackstabActive = on
        if on then
            if not State.KnifeItem then
                task.spawn(function()
                    local ki = _absGetKnifeItem()
                    if ki then _absInstallHooks(ki) end
                    _absStartLock()
                end)
            else
                _absStartLock()
            end
        else
            _absStopLock()
        end
    end

    getgenv().__AlwaysBackstab = State
    return State
end)()

        randomangle = false,

        fakelagstuds = 4,
        desync = false,
        desyncstuds = 3,
        microjitter = false,
        microstrength = 25,
        velocitybreaker = false,
    }

    local statemanager = {
        lastupdate = tick(),
        invertstate = false,
        smoothyaw = 0,
        smoothpitch = 0,
        smoothroll = 0,
        framecounter = 0
    }

    local utils = {
        getrandominrange = function(min, max)
            return min + math.random() * (max - min)
        end
    }

    local antiaim = {}

    function antiaim.calculateyaw(deltatime)
        local yaw = 0
        local currenttime = tick()
        if settings.yawtype == "jitter" then
            local minangle = math.rad(settings.minangle)
            local maxangle = math.rad(settings.maxangle)
            if settings.randomangle then
                yaw = utils.getrandominrange(-maxangle, maxangle)
            else
                yaw = math.random() > 0.5 and minangle or -minangle
            end
        elseif settings.yawtype == "spinbot" then
            local speed = utils.getrandominrange(
                settings.minspeed / 10,
                settings.maxspeed / 10
            )
            yaw = (currenttime * speed) % (2 * math.pi)
        elseif settings.yawtype == "random" then
            if statemanager.framecounter % 30 == 0 then
                yaw = utils.getrandominrange(
                    -math.rad(settings.maxangle),
                    math.rad(settings.maxangle)
                )
            else
                yaw = statemanager.smoothyaw
            end
        end
        statemanager.smoothyaw = yaw
        return yaw
    end

    function antiaim.calculatepitch()
        local pitch = 0
        if settings.pitchtype == "jitter" then
            local minangle = math.rad(settings.minangle)
            local maxangle = math.rad(settings.maxangle)
            if settings.randomangle then
                pitch = utils.getrandominrange(-maxangle, maxangle)
            else
                pitch = math.random() > 0.5 and minangle or -minangle
            end
        elseif settings.pitchtype == "spinbot" then
            pitch = math.sin(tick() * (settings.maxspeed / 10)) * math.rad(settings.maxangle)
        elseif settings.pitchtype == "random" then
            if statemanager.framecounter % 20 == 0 then
                pitch = utils.getrandominrange(math.rad(-89), math.rad(89))
            else
                pitch = statemanager.smoothpitch
            end
        end
        statemanager.smoothpitch = pitch
        return pitch
    end

    function antiaim.calculateroll()
        local roll = 0
        if settings.angletype == "tilt 45" then
            roll = math.rad(45)
        elseif settings.angletype == "tilt 90" then
            roll = math.rad(90)
        elseif settings.angletype == "upside down" then
            roll = math.rad(180)
        elseif settings.angletype == "custom" then
            roll = math.rad(settings.customangle)
        end
        statemanager.smoothroll = roll
        return roll
    end

    local function updantiaim(deltatime)
        if not settings.enabled then return end
        if settings.yawtype == "none" and settings.pitchtype == "none" and settings.angletype == "none" then
            return
        end
        local character = LocalPlayer.Character
        if not character then return end
        local rootpart = character:FindFirstChild("HumanoidRootPart")
        if not rootpart then return end
        statemanager.framecounter = statemanager.framecounter + 1
        local calculatedyaw = antiaim.calculateyaw(deltatime)
        local calculatedpitch = antiaim.calculatepitch()
        local calculatedroll = antiaim.calculateroll()
        local rotationcframe = CFrame.Angles(calculatedpitch, calculatedyaw, calculatedroll)
        rootpart.CFrame = rootpart.CFrame * rotationcframe
    end

    local function flushAntiAimMovementState()
        statemanager.framecounter = 0
        statemanager.smoothyaw = 0
        statemanager.smoothpitch = 0
        statemanager.smoothroll = 0
        local char = LocalPlayer.Character
        local hrp = char and char:FindFirstChild("HumanoidRootPart")
        if hrp then
            pcall(function()
                hrp.AssemblyAngularVelocity = Vector3.zero
            end)
        end
    end

    getgenv().InstanceFlushMovementState = flushAntiAimMovementState

    local underground_enabled = false
    local underground_oldpos = nil
    local undergroundDepthOffset = -2

    local function getFloorBelowPosition(pos)
        local rayOrigin = pos
        local rayDirection = Vector3.new(0, -500, 0)
        local raycastParams = RaycastParams.new()
        raycastParams.FilterType = Enum.RaycastFilterType.Exclude
        local fighter = FighterControllerAA and FighterControllerAA.LocalFighter
        if fighter and fighter.Entity and fighter.Entity.RootPart then
            raycastParams.FilterDescendantsInstances = {fighter.Entity.RootPart.Parent}
        end
        local result = workspace:Raycast(rayOrigin, rayDirection, raycastParams)
        if result then
            return CFrame.new(Vector3.new(pos.X, result.Position.Y + undergroundDepthOffset, pos.Z))
        end
        return nil
    end

    if CameraControllerAA and CameraControllerAA.Update then
        local oldUpdate = CameraControllerAA.Update
        CameraControllerAA.Update = function(...)
            if underground_enabled and FighterControllerAA and FighterControllerAA.LocalFighter
                and FighterControllerAA.LocalFighter.Entity and FighterControllerAA.LocalFighter.Entity.RootPart
                and underground_oldpos then
                FighterControllerAA.LocalFighter.Entity.RootPart.CFrame = underground_oldpos
            end
            return oldUpdate(...)
        end
    end

    local undergroundHeartbeatConn = nil
    local function startUndergroundHeartbeat()
        if undergroundHeartbeatConn then return end
        undergroundHeartbeatConn = game:GetService("RunService").Heartbeat:Connect(function()
            if not underground_enabled then
                underground_oldpos = nil
                return
            end
            local fighter = FighterControllerAA and FighterControllerAA.LocalFighter
            if not fighter or not fighter.Entity or not fighter.Entity.RootPart then
                underground_oldpos = nil
                return
            end
            underground_oldpos = fighter.Entity.RootPart.CFrame
            local currentPos = fighter.Entity.RootPart.Position
            local floorCFrame = getFloorBelowPosition(currentPos)
            if floorCFrame then
                fighter.Entity.RootPart.CFrame = floorCFrame
            end
        end)
    end

    local _antiAimConn = nil

    function _G.StartAntiAimExt()
        if not settings.enabled then
            settings.enabled = true
        end
        if not _antiAimConn then
            _antiAimConn = game:GetService("RunService").Heartbeat:Connect(updantiaim)
        end
    end

    function _G.StopAntiAimExt()
        settings.enabled = false
        if _antiAimConn then
            _antiAimConn:Disconnect()
            _antiAimConn = nil
        end
        flushAntiAimMovementState()
    end

    function _G.SetUndergroundExt(state)
        underground_enabled = state
        getgenv().InstanceUndergroundEnabled = state
        if state then
            startUndergroundHeartbeat()
        else
            underground_oldpos = nil
            if undergroundHeartbeatConn then
                undergroundHeartbeatConn:Disconnect()
                undergroundHeartbeatConn = nil
            end
        end
    end

    function _G.SetUndergroundDepthExt(depth)
        undergroundDepthOffset = depth
    end

    _G.AntiAimSettings = settings
end

local SlideBoostModule = (function()
    local Players = game:GetService("Players")
    local RunService = game:GetService("RunService")
    local LocalPlayer = Players.LocalPlayer

    _G.Features = _G.Features or {}
    _G.Features.SlideBoost = _G.Features.SlideBoost or {
        Enabled = false,
        Speed = 300
    }

    local mech = nil
    local connection = nil

    local function getMechanicsController()
        local success, result = pcall(function()
            return require(LocalPlayer.PlayerScripts.Controllers.MechanicsController)
        end)
        if success then
            mech = result
        else
            mech = nil
        end
        return mech
    end

    local function startSlideBoost()
        if connection then
            connection:Disconnect()
            connection = nil
        end
        local function boostLoop()
            if not _G.Features.SlideBoost.Enabled then
                return
            end
            if not mech then
                getMechanicsController()
            end
            if mech and mech.IsSliding then
                pcall(function()
                    mech._sliding_velocity.Velocity = mech._sliding_velocity.Velocity.Unit * _G.Features.SlideBoost.Speed
                end)
            end
        end
        connection = RunService.RenderStepped:Connect(boostLoop)
    end

    local function stopSlideBoost()
        if connection then
            connection:Disconnect()
            connection = nil
        end
    end

    local function setSlideBoost(enabled, speed)
        _G.Features.SlideBoost.Enabled = enabled
        if speed then
            _G.Features.SlideBoost.Speed = speed
        end
        if enabled then
            startSlideBoost()
        else
            stopSlideBoost()
        end
    end

    getMechanicsController()
    if _G.Features.SlideBoost.Enabled then
        startSlideBoost()
    end

    LocalPlayer.CharacterAdded:Connect(function()
        task.wait(1)
        getMechanicsController()
        if _G.Features.SlideBoost.Enabled then
            startSlideBoost()
        end
    end)

    return {
        setSlideBoost = setSlideBoost,
        getEnabled = function() return _G.Features.SlideBoost.Enabled end,
        getSpeed = function() return _G.Features.SlideBoost.Speed end,
    }
end)()

local function isEnemy(player)
    return player ~= LocalPlayer and player:GetAttribute("TeamID") ~= LocalPlayer:GetAttribute("TeamID")
end

local function getClosestEnemy()
    local closest = nil
    local minDist = math.huge
    local char = LocalPlayer.Character
    if not char or not char:FindFirstChild("HumanoidRootPart") then return nil end
    local root = char.HumanoidRootPart
    for _, p in ipairs(Players:GetPlayers()) do
        if isEnemy(p) and p.Character and p.Character:FindFirstChild("HumanoidRootPart") and p.Character:FindFirstChild("Humanoid") and p.Character.Humanoid.Health > 0 then
            local dist = (root.Position - p.Character.HumanoidRootPart.Position).Magnitude
            if dist < minDist then
                minDist = dist
                closest = p.Character
            end
        end
    end
    return closest
end

local function isClearLineOfSight(origin, targetPos, targetCharacter)
    local direction = (targetPos - origin)
    local distance = direction.Magnitude
    if distance < 0.01 then return true end
    direction = direction.Unit
    local rayParams = RaycastParams.new()
    rayParams.FilterType = Enum.RaycastFilterType.Blacklist
    rayParams.FilterDescendantsInstances = {LocalPlayer.Character, targetCharacter}
    local raycastResult = Workspace:Raycast(origin, direction * distance, rayParams)
    return raycastResult == nil
end

        end
    end

local function startFist()
    if fistConn then fistConn:Disconnect() end
    task.spawn(function()
        local modules = ReplicatedStorage:FindFirstChild("Modules")
        local itemLib = modules and modules:FindFirstChild("ItemLibrary")
        if not itemLib then return end
        local success, library = pcall(require, itemLib)
        if not (success and type(library) == "table") then return end
        local items = library.Items or library.Weapons or library.Melee or library.Utilities
        if type(items) ~= "table" then return end
        local meleeCooldowns = {
            "Cooldown", "AttackCooldown", "UseDelay", "SpinCooldown",
            "HeavyAttackCooldown", "AbilityCooldown", "FireCooldown"
        }
        for name, item in pairs(items) do
            if type(item) == "table" then
                local itemName = item.Name or name
                local isMelee = (item.Type == "Melee" or item.Category == "Melee" or (item.Damage and not item.AmmoType))
                if isMelee then
                    for _, field in ipairs(meleeCooldowns) do
                        if item[field] ~= nil then item[field] = 0 end
                    end
                end
                if string.lower(itemName) == "scythe" then
                    if item.DashCooldown ~= nil then item.DashCooldown = 0
                    elseif item.AbilityCooldown ~= nil then item.AbilityCooldown = 0 end

local function updateMobileESP()
    if not S.espActive then
        hideAllESP()
        return
    end
    local myChar = LocalPlayer.Character
    if not myChar then hideAllESP(); return end
    local myRoot = myChar:FindFirstChild("HumanoidRootPart")
    if not myRoot then hideAllESP(); return end

    local cam = workspace.CurrentCamera
    local enemies = {}
    for _, player in ipairs(Players:GetPlayers()) do
        if player == LocalPlayer then continue end
        if player:GetAttribute("TeamID") == LocalPlayer:GetAttribute("TeamID") then continue end
        local char = player.Character
        if char then
            local root = char:FindFirstChild("HumanoidRootPart")
            local head = char:FindFirstChild("Head")
            local hum = char:FindFirstChildOfClass("Humanoid")
            if root and head and hum and hum.Health > 0 then
                table.insert(enemies, {Char = char, Root = root, Head = head, Hum = hum, Player = player})
            end
        end
    end

    local idx = 1
    for _, enemy in ipairs(enemies) do
        if idx > MAX_ESP then break end
        local head = enemy.Head
        local root = enemy.Root
        local hum = enemy.Hum
        local headPos, onScreen = cam:WorldToViewportPoint(head.Position)
        local rootPos = cam:WorldToViewportPoint(root.Position)

        if onScreen then
            local height = (root.Position - head.Position).Magnitude * 3.0
            local width = height * 1.0
            local boxX = headPos.X - width/2
            local boxY = headPos.Y

            local esp = espPool[idx]
            esp.box.Visible = true
            esp.box.Position = UDim2.new(0, boxX, 0, boxY)
            esp.box.Size = UDim2.new(0, width, 0, height)
            esp.outline.BorderColor3 = Color3.fromHSV(hum.Health/hum.MaxHealth*0.33, 1, 1)

            if S.espShowName then
                esp.name.Text = enemy.Player.Name
                esp.name.Position = UDim2.new(0.5,0,0,-16)
                esp.name.Visible = true
            else
                esp.name.Visible = false
            end

            if S.espShowDistance then
                esp.dist.Text = math.floor((myRoot.Position - root.Position).Magnitude) .. "m"
                esp.dist.Position = UDim2.new(0.5,0,1,9)
                esp.dist.Visible = true
            else
                esp.dist.Visible = false
            end

            if S.espShowHealthBar then
                esp.hpBar.Visible = true
                esp.hpFill.Size = UDim2.new(hum.Health/hum.MaxHealth, 1, 1, 0)
                esp.hpFill.BackgroundColor3 = Color3.fromHSV(hum.Health/hum.MaxHealth*0.33, 1, 1)
            else
                esp.hpBar.Visible = false
            end

            idx = idx + 1
        end
    end

    for i = idx, MAX_ESP do
        espPool[i].box.Visible = false
    end
end

local function toggleAimbot(on)
    S.aimbotActive = on
    if on then
        if aimbotConn then aimbotConn:Disconnect() end
        aimbotConn = RunService.RenderStepped:Connect(function(dt)
            if not S.aimbotActive then return end
            local targetPart = nil
            local camPos = Camera.CFrame.Position
            local camDir = Camera.CFrame.LookVector
            local closestAngle = S.aimFov
            for _, p in ipairs(Players:GetPlayers()) do
                if p == LocalPlayer then continue end
                if S.teamCheck and not isEnemy(p) then continue end
                local char = p.Character
                if char then
                    local part = char:FindFirstChild(S.aimTargetPart)
                    local hum = char:FindFirstChildWhichIsA("Humanoid")
                    if part and hum and hum.Health > 0 then
                        local toTarget = (part.Position - camPos).Unit
                        local angle = math.acos(math.clamp(camDir:Dot(toTarget), -1, 1))
                        local degrees = math.deg(angle)
                        if degrees <= closestAngle then
                            if isClearLineOfSight(camPos, part.Position, char) then
                                closestAngle = degrees
                                targetPart = part
                            end
                        end
                    end
                end
            end
            if targetPart then
                local targetCF = CFrame.lookAt(camPos, targetPart.Position)
                if S.aimSmooth <= 0.01 then
                    if CameraControllerAimbot and CameraControllerAimbot.MimicRotation then
                        pcall(function()
                            CameraControllerAimbot:MimicRotation(targetCF)
                        end)
                    else
                        Camera.CFrame = targetCF
                    end
                else
                    local smoothFactor = 1 - math.pow(0.001, dt / S.aimSmooth)
                    local newCF = Camera.CFrame:Lerp(targetCF, math.clamp(smoothFactor, 0, 1))
                    if CameraControllerAimbot and CameraControllerAimbot.MimicRotation then
                        pcall(function()
                            CameraControllerAimbot:MimicRotation(newCF)
                        end)
                    else
                        Camera.CFrame = newCF
                    end
                end
            end
        end)
    else
        if aimbotConn then aimbotConn:Disconnect(); aimbotConn = nil end
    end
end

local function triggerbot_IsADS()
    local char = LocalPlayer.Character
    if not char then return false end
    local tool = char:FindFirstChildOfClass("Tool")
    if tool and tool:FindFirstChild("GunMixin") then
        local gun = require(tool.GunMixin)
        return gun.IsAiming
    end
    pcall(function()
        local fc = require(LocalPlayer.PlayerScripts.Controllers.FighterController)
        local fighter = fc:GetFighter(LocalPlayer)
        if fighter and fighter.IsAiming then
            return fighter:IsAiming()
        end
    end)
    return false
end

local function triggerbot_GetTarget()
    local camPos = Camera.CFrame.Position
    local camDir = Camera.CFrame.LookVector
    local closestAngle = S.silentFov
    local targetPart = nil
    for _, p in ipairs(Players:GetPlayers()) do
        if p == LocalPlayer then continue end
        if S.triggerbotTeamCheck and p:GetAttribute("TeamID") == LocalPlayer:GetAttribute("TeamID") then continue end
        local char = p.Character
        if char then
            local head = char:FindFirstChild("Head")
            local hum = char:FindFirstChildOfClass("Humanoid")
            if head and hum and hum.Health > 0 then
                local toTarget = (head.Position - camPos).Unit
                local angle = math.acos(math.clamp(camDir:Dot(toTarget), -1, 1))
                local degrees = math.deg(angle)
                if degrees <= closestAngle then
                    if not S.triggerbotWallCheck or isClearLineOfSight(camPos, head.Position, char) then
                        closestAngle = degrees
                        targetPart = head
                    end
                end
            end
        end
    end
    return targetPart
end

local function triggerbot_Shoot()
    local char = LocalPlayer.Character
    if not char then return end
    local tool = char:FindFirstChildOfClass("Tool")
    if not tool then return end
    pcall(function()
        local remote = ReplicatedStorage:FindFirstChild("Remotes"):FindFirstChild("Replication"):FindFirstChild("Fighter"):FindFirstChild("FireServer")
        if remote then
            remote:FireServer()
        else
            VirtualInputManager:SendMouseButtonEvent(
                ViewportSize.X/2, ViewportSize.Y/2,
                0, true, game, 1
            )
            task.wait(0.05)
            VirtualInputManager:SendMouseButtonEvent(
                ViewportSize.X/2, ViewportSize.Y/2,
                0, false, game, 1
            )
        end
    end)
end

local function triggerbotUpdate()
    if not S.triggerbotEnabled then return end
    if not UserInputService:IsKeyDown(S.triggerbotKey) then
        S.triggerbotActive = false
        return
    end
    if S.triggerbotOnlyADS and not triggerbot_IsADS() then return end
    local target = triggerbot_GetTarget()
    if target then
        local now = tick()
        local delay = S.triggerbotDelay / 1000
        if now - S.triggerbotLastShot >= delay then
            S.triggerbotLastShot = now
            triggerbot_Shoot()
        end
    end
end

local velHistory={}
local function _rbRecVel(c,r)
    local now,pos=tick(),r.Position
    if not velHistory[c] then velHistory[c]={{pos=pos,t=now},{pos=pos,t=now}}
    else velHistory[c][1]=velHistory[c][2]; velHistory[c][2]={pos=pos,t=now} end
end
local function _rbPredict(c,r,lead)
    local h=velHistory[c]; if not h then return r.Position end
    local dt=h[2].t-h[1].t; if dt<0.001 then return r.Position end
    return r.Position+((h[2].pos-h[1].pos)/dt)*lead
end

            if FlickCfg.PredictionOn and FlickCfg.PredictionScale>0 then
                fin=base+(_rbPredict(fp.Character,fr,FlickCfg.PredictionScale)-fr.Position)
            end
            local predHead=FlickCfg.PredictionOn and (fh.Position+(_rbPredict(fp.Character,fr,FlickCfg.PredictionScale)-fr.Position)) or fh.Position
            flickDesync=CFrame.lookAt(fin,predHead)

local srCache,srT=false,0
local function _rbIsShootingRange()
    local now=os.clock()
    if now-srT<0.5 then return srCache end
    srT=now
    local function m(v)
        if v==nil then return false end
        local t=tostring(v):lower():gsub("[%s_%-]","")
        return t:find("shootingrange",1,true) or t:find("firingrange",1,true) or t:find("사격장",1,true)
    end
    for _,o in ipairs({_rbWS,_rbPlayer}) do
        for _,a in ipairs({"Map","MapName","Mode","GameMode","Arena","Environment","EnvironmentName","ShootingRange"}) do
            local v=o:GetAttribute(a)
            if (v==true and a=="ShootingRange") or m(v) then srCache=true; return true end
        end
    end
    local c=_rbPlayer.Character; local cur=c
    while cur and cur~=_rbWS do
        if m(cur.Name) then srCache=true; return true end
        cur=cur.Parent
    end
    local root=c and c:FindFirstChild("HumanoidRootPart")
    local function near(o)
        if not m(o.Name) or not root then return false end
        local p=o:IsA("BasePart") and o or o:FindFirstChildWhichIsA("BasePart",true)
        return p and (p.Position-root.Position).Magnitude<2000
    end
    for _,ch in ipairs(_rbWS:GetChildren()) do
        if near(ch) then srCache=true; return true end
        for _,g in ipairs(ch:GetChildren()) do if near(g) then srCache=true; return true end end
    end
    srCache=false; return false
end

                if RagebotSettings.priorityVoided and tr.Position.Magnitude>1e6 then v=v-2e9 end
                if RagebotSettings.priorityAttackers and _rbScanWeapon(p)~="" then v=v-1e9 end
            end
            if v<bv then bv=v; best=p end
        end
    end
    return best
end

    local c=_rbPlayer.Character; local r=_rbRoot(c); local h=c and c:FindFirstChildOfClass("Humanoid")

    return _rbFighterFn()~=nil
end

local function _rbBuildCam(from,part)
    if not _rbUtil or not part then return nil end
    local look=CFrame.new(from,part.Position)
    local d={}
    d[utf8.char(1)]={
        [utf8.char(0)]=_rbUtil:EncodeCFrame(look),
        [utf8.char(1)]=_rbUtil:EncodeCFrame(look),
        [utf8.char(2)]=part,
        [utf8.char(3)]=_rbUtil:EncodeCFrame(part.CFrame:ToObjectSpace(CFrame.new(part.Position))),
    }
    return d
end
local function _rbUpdCam() if rbUpdateCamRemote then return rbUpdateCamRemote end
    local r=_rbRS:FindFirstChild("Remotes"); r=r and r:FindFirstChild("Replication"); r=r and r:FindFirstChild("Fighter"); r=r and r:FindFirstChild("UpdateCameraRotation")

local function _rbMelee(st,head)
    if not _rbUtil or not _rbEnum then return false end
    local f=_rbFighterFn(); local item=f and f.EquippedItem; if not item then return false end
    local now=tick()
    local ad=math.max((item.Info and item.Info.AttackDelay) or 0,0.05)
    if now-rbLastMeleeFire<ad then return false end
    local tc=st.target and st.target.Character; local tr=tc and _rbRoot(tc); if not tr then return false end
    local my=_rbRoot(_rbPlayer.Character); if not my then return false end
    local dx,dz=tr.Position.X-my.Position.X,tr.Position.Z-my.Position.Z
    local mag=math.sqrt(dx*dx+dz*dz); if mag<0.01 then return false end
    dx,dz=dx/mag,dz/mag
    local yaw=math.atan2(-dx,-dz)+math.pi
    local ok,enc=pcall(function() return _rbUtil:EncodeCameraRotation(Vector2.new(0,yaw)) end)
    if not ok or not enc then return false end
    local ucr=_rbUpdCam()
    local use=_rbRS.Remotes.Replication.Fighter.UseItem
    local oid=item.Get and item:Get("ObjectID")
    local se=_rbEnum:ToEnum("StartShooting")
    local n=math.max(1,math.floor(RagebotSettings.shootAttempts or 1))
    rbLastMeleeFire=now
    local fired=false
    for _=1,n do
        if ucr then pcall(function() ucr:FireServer(enc,nil) end) end
        if oid and se and use then local ok2=pcall(function() use:FireServer(oid,se,nil,nil) end); if ok2 then fired=true end end
    end
    return fired
end
local function _rbDoFire(st,part)
    if st.weaponKind=="melee" then return _rbMelee(st,part) end
    local f=_rbFighterFn(); local item=f and f.EquippedItem
    if not item or not part then return false end
    local cam=_rbWS.CurrentCamera
    local from=(st.csyncCF and st.csyncCF.Position) or (cam and cam.CFrame.Position) or part.Position
    local any=false
    local n=math.max(1,math.floor(RagebotSettings.shootAttempts or 1))
    local use=_rbRS.Remotes.Replication.Fighter.UseItem
    for _=1,n do
        local fired=false
        if RagebotSettings.useManipulation and use and _rbEnum and _rbUtil then
            local ammo=item.Get and (item:Get("Ammo") or 0) or 0
            if ammo>0 then
                local oid=item:Get("ObjectID"); local se=_rbEnum:ToEnum("StartShooting")
                local data=_rbBuildCam(from,part)
                if oid and se and data then fired=pcall(function() use:FireServer(oid,se,data,nil) end) end
            end
        end
        if not fired and item.UseItem then fired=pcall(function() item:UseItem() end) end
        if not fired and f and f.UseItem then fired=pcall(function() f:UseItem() end) end
        any=any or fired
    end
    return any
end
local function _rbAmmo(st)
    local f=_rbFighterFn(); local item=f and f.EquippedItem
    if not f or not item then return false end
    local slot=item:Get("Slot") or 1
    if slot==(RagebotSettings.meleeSlot or 3) then return false end
    local ammo=item:Get("Ammo") or 0
    local now=tick()
    if f:Get("Reloading") then
        st.hideOrbitUntil=math.max(st.hideOrbitUntil or 0,now+0.25)
        st.ammoActionAt=math.max(st.ammoActionAt or 0,now+0.1)
        return true
    end
    if ammo>0 then return false end
    if now<(st.ammoActionAt or 0) then return true end
    local prim=RagebotSettings.primarySlot or 1
    local sec=RagebotSettings.secondarySlot or 2
    if slot==prim and RagebotSettings.autoSwapSecondary then
        st.ammoActionAt=now+0.45; st.hideOrbitUntil=math.max(st.hideOrbitUntil or 0,now+0.45)
        st.ammoFallbackSlot=sec; pcall(_rbPressKey,rbSlotKey[sec] or Enum.KeyCode.Two); return true
    end
    if slot==sec and RagebotSettings.autoReloadPrimary then
        st.ammoActionAt=now+0.8; st.hideOrbitUntil=math.max(st.hideOrbitUntil or 0,now+0.9)
        st.ammoFallbackSlot=prim; pcall(_rbPressKey,rbSlotKey[prim] or Enum.KeyCode.One)
        task.delay(0.18,function()
            if not st.active then return end
            local f2=_rbFighterFn(); local i2=f2 and f2.EquippedItem
            if f2 and i2 and (i2:Get("Slot") or 1)==prim and (i2:Get("Ammo") or 0)<=0 and not f2:Get("Reloading") then
                pcall(_rbPressKey,Enum.KeyCode.R)
            end
        end)
        return true
    end
    if slot==prim and RagebotSettings.autoReloadPrimary then
        st.ammoActionAt=now+0.8; st.hideOrbitUntil=math.max(st.hideOrbitUntil or 0,now+0.9)
        pcall(_rbPressKey,Enum.KeyCode.R); return true
    end
    return true
end
local function _rbEquipSlot(s) s=tonumber(s) or 1; pcall(_rbPressKey,rbSlotKey[s] or Enum.KeyCode.One) end
local function _rbWeaponProfile(st)
    if RagebotSettings.weaponSpecialize==false then return "default" end
    local f=_rbFighterFn(); local item=f and f.EquippedItem
    local slot=item and item:Get("Slot") or nil
    local pref=RagebotSettings.preferredWeapon or "primary"
    if RagebotSettings.autoEquipPreferred~=false and not st.preferredEquipped then
        local want=(pref=="secondary" and (RagebotSettings.secondarySlot or 2)) or (pref=="melee" and (RagebotSettings.meleeSlot or 3)) or (RagebotSettings.primarySlot or 1)
        if slot~=want then _rbEquipSlot(want); slot=want end
        st.preferredEquipped=true
    end
    local kind
    if slot==(RagebotSettings.meleeSlot or 3) or pref=="melee" then kind="melee"
    elseif slot==(RagebotSettings.secondarySlot or 2) or pref=="secondary" then kind="secondary"
    else kind="primary" end
    if kind=="primary" then
        RagebotSettings.hyper=true; RagebotSettings.orbitDist=3.2; RagebotSettings.orbitHeight=2.2
        RagebotSettings.strafeSpeed=5e6; RagebotSettings.teleportDelay=0; RagebotSettings.behindDist=3.5
        RagebotSettings.randomMovement=false; RagebotSettings.dirBack=true; RagebotSettings.dirFront=false
        RagebotSettings.dirLeft=true; RagebotSettings.dirRight=true
    elseif kind=="secondary" then
        RagebotSettings.hyper=false; RagebotSettings.orbitDist=2.6; RagebotSettings.orbitHeight=1.6
        RagebotSettings.strafeSpeed=4e6; RagebotSettings.teleportDelay=0; RagebotSettings.behindDist=3.0
        RagebotSettings.randomMovement=true; RagebotSettings.randomRefresh=0.07
        RagebotSettings.dirBack=true; RagebotSettings.dirFront=true; RagebotSettings.dirLeft=true; RagebotSettings.dirRight=true
    else
        RagebotSettings.hyper=true; RagebotSettings.orbitDist=0.1; RagebotSettings.orbitHeight=-0.5
        RagebotSettings.strafeSpeed=999999; RagebotSettings.teleportDelay=0; RagebotSettings.behindDist=0.5
        RagebotSettings.undergroundDepth=8; RagebotSettings.randomMovement=false
        RagebotSettings.dirBack=true; RagebotSettings.dirFront=false; RagebotSettings.dirLeft=false
        RagebotSettings.dirRight=false; RagebotSettings.dirDown=true
        RagebotSettings.undergroundBounce=true; RagebotSettings.undergroundBounceSpeed=999999
    end
    st.weaponKind=kind; return kind
end

end

local function _rbStartCsync(st,gen)
    if st.csyncHbConn then return end
    st.csyncHbConn=_rbRun.Heartbeat:Connect(function()
        local root=_rbRoot(_rbPlayer.Character); if not root then return end
        if st.csyncWroteFake and st.csyncLocalCF then _rbRestoreRoot(st,root) end
        if _rbSettling() then
            st.csyncLocalCF=root.CFrame; st.csyncLocalLV=root.AssemblyLinearVelocity; st.csyncLocalAV=root.AssemblyAngularVelocity
            st.csyncWroteFake=false; return
        end
        st.csyncLocalCF=root.CFrame; st.csyncLocalLV=root.AssemblyLinearVelocity; st.csyncLocalAV=root.AssemblyAngularVelocity
        if st.csyncCF then
            root.CFrame=st.csyncCF
            local fv=st.csyncLV or st.csyncLocalLV or root.AssemblyLinearVelocity
            local lv=st.csyncLocalLV or root.AssemblyLinearVelocity
            root.AssemblyLinearVelocity=Vector3.new(fv.X,lv.Y,fv.Z)
            root.AssemblyAngularVelocity=st.csyncAV or st.csyncLocalAV or root.AssemblyAngularVelocity
            st.csyncWroteFake=true
        else st.csyncWroteFake=false end
    end)
    _rbRun:BindToRenderStep("RB_Csync_"..gen,Enum.RenderPriority.Camera.Value-1,function()
        local root=_rbRoot(_rbPlayer.Character)
        if not root or not st.csyncLocalCF then return end
        if st.csyncWroteFake and _rbRestoreRoot(st,root) then st.csyncWroteFake=false end
    end)
end
local function _rbStopCsync(st,gen)
    if st.csyncHbConn then st.csyncHbConn:Disconnect(); st.csyncHbConn=nil end
    pcall(function() _rbRun:UnbindFromRenderStep("RB_Csync_"..gen) end)
    _rbRestoreRoot(st,_rbRoot(_rbPlayer.Character))
    _rbClearCsync(st)
    st.csyncLocalCF=nil; st.csyncLocalLV=nil; st.csyncLocalAV=nil; st.csyncWroteFake=false
end
local function _rbEnterVoid(st)
    st.voidTargetCF=nil; st.voidExposed=false; st.orbitClientCF=nil
    if not st.active or not RagebotSettings.on then _rbClearCsync(st); _rbUpdateStatus(st); return end
    if RagebotSettings.voidSpam then _rbSetVoidCsync(st,_rbVoidRandCF()) else _rbClearCsync(st) end
    _rbUpdateStatus(st)
end

    st.voidHbConn=_rbRun.Heartbeat:Connect(function()

        elseif RagebotSettings.voidSpam then _rbSetVoidCsync(st,_rbVoidRandCF())
        else _rbClearCsync(st) end
    end)
end
local function _rbDisableVoidCsync(st,gen)

    st.voidTargetCF=nil; st.voidThread=nil; st.voidExposed=false
end

local function _rbStartVoidLoop(st,gen)
    if st.voidThread then return end
    _rbEnableVoidCsync(st,gen)
    local thread
    thread=task.spawn(function()
        while st.active and RagebotSettings.on and rbGen==gen and not st.suspended do
            if _rbSettling() then st.voidTargetCF=nil; st.voidExposed=false; _rbClearCsync(st); task.wait(0.03); continue end
            if not _rbInMatch() or not _rbHasTgt(st) or _rbKatana(st.target) then _rbEnterVoid(st); task.wait(0.1); continue end
            _rbEnterVoid(st)
            if RagebotSettings.voidHideTime>0 then task.wait(RagebotSettings.voidHideTime) end
            if not st.active or not RagebotSettings.on or rbGen~=gen or st.suspended or not _rbInMatch() then break end
            local t=st.target
            if _rbHasTgt(st) and not _rbKatana(t) then
                local tc=t.Character; local tr=_rbRoot(tc); local head=tc and (tc:FindFirstChild("Head") or tr)
                if tr and head then
                    local sp=_rbRiot(t) and (tr.Position-tr.CFrame.LookVector*(RagebotSettings.behindDist or 4)) or (tr.Position-tr.CFrame.LookVector*2.5+Vector3.new(0,1.5,0))
                    if not _rbSafe(sp,t) then _rbEnterVoid(st); task.wait(0.1); continue end
                    local cf=CFrame.new(sp,head.Position)
                    st.voidExposed=true; st.voidTargetCF=cf
                    _rbSetVoidCsync(st,cf,Vector3.zero,Vector3.zero)
                    _rbUpdateStatus(st)
                    if RagebotSettings.voidShootTime>0 then task.wait(RagebotSettings.voidShootTime) end
                    if _rbHasTgt(st) and not _rbKatana(t) then _rbDoFire(st,head) end
                    task.wait(0.05); _rbEnterVoid(st)
                end
            end
        end
        if st.voidThread==thread then st.voidThread=nil end
        if rbGen==gen and not st.suspended and st.voidThread==nil then _rbDisableVoidCsync(st,gen) end
    end)
    st.voidThread=thread
end
local function _rbNoclip(st)
    if st.noclipConn then return end
    st.noclipConn=_rbRun.Stepped:Connect(function()
        local c=_rbPlayer.Character; if not c then return end
        for _,p in c:GetDescendants() do if p:IsA("BasePart") then p.CanCollide=false end end
    end)
end
local function _rbStartAmmoLoop(st)
    if st.ammoThread then return end
    st.ammoThread=task.spawn(function()
        while st.active do
            if _rbIsShootingRange() then task.wait(0.1); continue end
            if not _rbAmmo(st) and _rbShouldShoot(st) and not RagebotSettings.hyper then
                local tc=st.target and st.target.Character
                local head=tc and (tc:FindFirstChild("Head") or _rbRoot(tc))
                if head then
                    if RagebotSettings.shootDelay>0 then task.wait(RagebotSettings.shootDelay) end
                    _rbDoFire(st,head)
                end
            end
            task.wait(RagebotSettings.acSpd)
        end
        st.ammoThread=nil
    end)
end
local function _rbDirs(tr)
    local d={}; local l=tr.CFrame.LookVector; local r=tr.CFrame.RightVector
    if RagebotSettings.dirBack then table.insert(d,-l) end
    if RagebotSettings.dirFront then table.insert(d,l) end
    if RagebotSettings.dirLeft then table.insert(d,-r) end
    if RagebotSettings.dirRight then table.insert(d,r) end
    if #d==0 then d[1]=-l; d[2]=r; d[3]=-r end
    return d
end
local function _rbPickOffset(tr,head)
    local ds=_rbDirs(tr); local dir=ds[math.random(1,#ds)]
    local rad=math.clamp(RagebotSettings.orbitDist or 3,1.25,5)
    local h=math.clamp(RagebotSettings.orbitHeight or 2,-2,6)
    local pos=head.Position+dir*rad+Vector3.new(0,h,0)
    if RagebotSettings.dirUp and math.random()<0.2 then pos=pos+Vector3.new(0,math.max(1,h),0)
    elseif RagebotSettings.dirDown and math.random()<0.15 then pos=pos+Vector3.new(0,-math.max(1,math.min(3,RagebotSettings.undergroundDepth or 2)),0) end
    return pos
end
local function _rbUGPos(head,tr)
    local d=math.clamp(RagebotSettings.undergroundDepth or 6,3,8)
    local r=math.clamp(RagebotSettings.orbitDist or 3,1.25,4)
    local base=head.Position-tr.CFrame.LookVector*r
    if RagebotSettings.undergroundBounce then
        local sp=math.max(1,RagebotSettings.undergroundBounceSpeed or 5000)
        local ph=math.floor(os.clock()*sp)%2
        if ph==0 then return base+Vector3.new(0,-d,0) end
        return base+Vector3.new(0,RagebotSettings.undergroundSurfaceHeight or 0,0)
    end
    return base+Vector3.new(0,-d,0)
end
local function _rbStop()
    rbGen=rbGen+1
    if rbActiveState then
        local st=rbActiveState
        st.active=false; RagebotSettings.on=false
        _rbSetStatus(false,nil,true)
        if st.conn then st.conn:Disconnect(); st.conn=nil end
        if st.noclipConn then st.noclipConn:Disconnect(); st.noclipConn=nil end
        st.target=nil; st.voidExposed=false; st.nextTeleportAt=0; st.ammoActionAt=0
        st.hideOrbitUntil=0; st.randPos=nil; st.randT=0; st.lastFakePos=nil
        rbInMatchT=0; rbInMatch=false
        _rbStopCsync(st,rbGen-1); _rbDisableVoidCsync(st,rbGen-1); _rbStopOrbitRender(st,rbGen-1)
        local c=_rbPlayer.Character
        if c then for _,p in c:GetDescendants() do if p:IsA("BasePart") then p.CanCollide=true end end end
        rbActiveState=nil
    end
    if rbCharConn then rbCharConn:Disconnect(); rbCharConn=nil end
end
local function _rbStart()
    if rbActiveState then _rbStop() end
    RagebotSettings.on=true; RagebotSettings.settleUntil=0
    rbGen=rbGen+1; local gen=rbGen
    local st={active=true,target=nil,conn=nil,ammoThread=nil,voidThread=nil,voidHbConn=nil,csyncHbConn=nil,
        voidExposed=false,voidTargetCF=nil,nextTeleportAt=0,ammoActionAt=0,ammoFallbackSlot=nil,
        hideOrbitUntil=0,preferredEquipped=false,randPos=nil,randT=0,lastFakePos=nil,
        csyncCF=nil,csyncLV=nil,csyncAV=nil,csyncLocalCF=nil,csyncLocalLV=nil,csyncLocalAV=nil,
        csyncWroteFake=false,noclipConn=nil,suspended=_rbIsShootingRange(),weaponKind=nil,
        orbitRenderRunning=false,orbitClientCF=nil}
    rbActiveState=st
    _rbSetStatus(true,nil,true)
    _rbStartAmmoLoop(st)
    if not st.suspended then
        _rbNoclip(st)
        local m=RagebotSettings.mode
        if m=="Void" then _rbStartVoidLoop(st,gen)
        elseif m=="Orbit" then _rbEnableVoidCsync(st,gen); _rbStartOrbitRender(st,gen)
        else _rbStartCsync(st,gen) end
    end
    local aaPhase,orbitAngle=0,math.random()*math.pi*2
    local function rndDir()
        local a=math.random()*math.pi*2
        return Vector3.new(math.cos(a),0,math.sin(a))
    end
    st.conn=_rbRun.Stepped:Connect(function(_,dt)
        if not st.active or not RagebotSettings.on then
            if st.conn then st.conn:Disconnect(); st.conn=nil end; return
        end
        if _rbIsShootingRange() then
            if not st.suspended then
                st.suspended=true; st.target=nil; st.randPos=nil; st.voidTargetCF=nil; st.voidExposed=false
                if st.noclipConn then st.noclipConn:Disconnect(); st.noclipConn=nil end
                _rbStopCsync(st,gen); _rbDisableVoidCsync(st,gen); _rbStopOrbitRender(st,gen)
                _rbUpdateStatus(st)
            end
            return
        end
        if st.suspended then
            st.suspended=false; _rbNoclip(st)
            local m=RagebotSettings.mode
            if m=="Void" then _rbStartVoidLoop(st,gen)
            elseif m=="Orbit" then _rbEnableVoidCsync(st,gen); _rbStartOrbitRender(st,gen)
            else _rbStartCsync(st,gen) end
        end
        local root=_rbRoot(_rbPlayer.Character); if not root then return end
        if _rbSettling() then st.voidTargetCF=nil; st.voidExposed=false; st.orbitClientCF=nil; _rbClearCsync(st); _rbUpdateStatus(st); return end
        if not _rbInMatch() then
            _rbClearCsync(st); st.target=nil
            if RagebotSettings.mode=="Orbit" then _rbEnterVoid(st) else _rbUpdateStatus(st) end
            return
        end
        local now=tick()
        if st.target and _rbDead(st.target) then st.target=nil end
        if st.target and _rbInvincible(st.target) then
            _rbClearCsync(st)
            if RagebotSettings.mode=="Orbit" then _rbEnterVoid(st) else _rbUpdateStatus(st) end
            return
        end
        if now-rbTgtT>=0.0005 and (RagebotSettings.autoSwitch or not st.target) then
            rbTgtT=now
            if RagebotSettings.autoSwitch then
                local t=_rbBestTarget()
                if t then
                    if RagebotSettings.sendNotification and st.target~=t then
                        task.spawn(function() Library:Notify("ragebot → "..t.Name,"success") end)
                    end
                    st.target=t
                end
            elseif not st.target then st.target=_rbBestTarget() end
        end
        if not st.target then
            _rbClearCsync(st)
            if RagebotSettings.mode=="Orbit" then _rbEnterVoid(st) else _rbUpdateStatus(st) end
            return
        end
        local tc=st.target.Character; local tr=_rbRoot(tc); local head=tc and (tc:FindFirstChild("Head") or tr)
        if not tc or not tr or not head then st.target=nil; _rbUpdateStatus(st); return end
        if _rbNearOther(tr.Position,st.target) then
            st.target=nil; st.randPos=nil
            if RagebotSettings.mode=="Orbit" or RagebotSettings.mode=="Void" then _rbEnterVoid(st)
            else _rbClearCsync(st); _rbUpdateStatus(st) end
            return
        end
        _rbUpdateStatus(st); _rbWeaponProfile(st)
        if RagebotSettings.mode=="Void" then return end
        if RagebotSettings.mode=="Orbit" and (now<(st.hideOrbitUntil or 0) or _rbAmmo(st)) then
            st.voidTargetCF=nil; st.voidExposed=false; st.orbitClientCF=nil; _rbEnterVoid(st); return
        end
        local isUG=RagebotSettings.mode=="Underground" or RagebotSettings.undergroundOverride==true
        local isSh=_rbRiot(st.target)
        local height=math.clamp(RagebotSettings.orbitHeight or 2,-2,6)
        local radius=math.clamp(RagebotSettings.orbitDist or 3,1.25,5)
        local tPos
        if isSh then tPos=tr.Position-tr.CFrame.LookVector*(RagebotSettings.behindDist or 3)
        elseif isUG then tPos=_rbUGPos(head,tr)
        elseif RagebotSettings.mode=="Teleport" then
            if RagebotSettings.randomMovement then
                if not st.randPos or (now-(st.randT or 0))>=(RagebotSettings.randomRefresh or 0.08) then
                    st.randT=now
                    st.randPos=_rbPickOffset(tr,head)+rndDir()*(math.random()*1.05)+Vector3.new(0,(math.random()*2-1)*0.7,0)
                end
                tPos=st.randPos
            else tPos=_rbPickOffset(tr,head) end
        elseif RagebotSettings.mode=="Orbit" then
            orbitAngle=orbitAngle+dt*math.max(1,(RagebotSettings.strafeSpeed or 5)*1500)
            tPos=head.Position+Vector3.new(math.cos(orbitAngle)*radius,height,math.sin(orbitAngle)*radius)
        else tPos=_rbUGPos(head,tr) end
        if not _rbSafe(tPos,st.target) then
            st.randPos=nil
            if RagebotSettings.mode=="Orbit" and not isUG then
                st.voidTargetCF=nil; st.voidExposed=false; st.orbitClientCF=nil; _rbEnterVoid(st)
            else _rbClearCsync(st); _rbUpdateStatus(st) end
            return
        end
        local face=CFrame.new(tPos,head.Position)
        if RagebotSettings.antiAim then
            aaPhase=aaPhase+dt*20
            face=CFrame.new(tPos,head.Position)*CFrame.Angles(0,math.rad(math.sin(aaPhase)*70),0)
        end
        if RagebotSettings.mode=="Orbit" and not isUG then
            if RagebotSettings.hyper then
                st.voidExposed=true; st.voidTargetCF=face; st.orbitClientCF=face
                _rbSetCsync(st,face,tPos,dt); _rbUpdateStatus(st)
                if _rbShouldShoot(st) then _rbDoFire(st,head) end
            end
        else
            _rbSetCsync(st,face,tPos,dt)
            if RagebotSettings.hyper then
                if _rbShouldShoot(st) then _rbDoFire(st,head) end
            elseif RagebotSettings.mode=="Teleport" and not isUG then
                if now>=(st.nextTeleportAt or 0) then
                    st.nextTeleportAt=now+math.max(0.01,RagebotSettings.teleportDelay or 0.04)
                    if _rbShouldShoot(st) then _rbDoFire(st,head) end
                end
            elseif isUG then
                if _rbShouldShoot(st) then _rbDoFire(st,head) end
            end
        end
    end)
    rbCharConn=_rbPlayer.CharacterAdded:Connect(function()
        _rbStopCsync(st,gen); _rbDisableVoidCsync(st,gen); _rbStopOrbitRender(st,gen)
        st.target=nil; _rbClearCsync(st)
        st.csyncLocalCF=nil; st.csyncLocalLV=nil; st.csyncLocalAV=nil; st.csyncWroteFake=false
        st.voidExposed=false; st.hideOrbitUntil=0; rbLastMeleeFire=0
        if st.active then
            task.wait(0.5)
            if st.active then
                local m=RagebotSettings.mode
                if m=="Void" then _rbStartVoidLoop(st,gen)
                elseif m=="Orbit" then _rbEnableVoidCsync(st,gen); _rbStartOrbitRender(st,gen)
                else _rbStartCsync(st,gen) end
            end
        end
    end)
end

    return math.atan2(-(dx/mag),-(dz/mag))

getgenv().ASCfg = {
    Enabled=false, FireRate=0.05, WeaponSlot="Primary", MaxDistance=500,
    FOVEnabled=false, FOVRadius=150, TargetLock=false, HitPart="Head",
    TargetPlayers=false, TeamCheck=true, VisibleOnly=false,
    PredictEnabled=false, PredictTime=0.07, Spread=0.02,
    BurstEnabled=false, BurstCount=3, BurstDelay=0.01,
}
local asCache,asCacheT={},0
local AS_TTL=0.35
local function _asRebuild()
    asCache={}
    local ch=_rbPlayer.Character
    for _,o in ipairs(_rbWS:GetDescendants()) do
        if o:IsA("Model") and o~=ch then
            local r=o:FindFirstChild("HumanoidRootPart")
            local h=o:FindFirstChild("Head")
            local t=o:FindFirstChild("UpperTorso") or o:FindFirstChild("Torso")
            local hum=o:FindFirstChildWhichIsA("Humanoid")
            if r and h and hum then
                local p=_rbPlayers:GetPlayerFromCharacter(o)
                table.insert(asCache,{Model=o,Root=r,Head=h,Torso=t,Hum=hum,IsPlayer=p~=nil,Player=p})
            end
        end
    end
end
local asSlots={Primary=1,Secondary=2,Melee=3}
local function _asSlot() return asSlots[getgenv().ASCfg.WeaponSlot] or 1 end
local function _asInFov(pos)
    local vp,on=_rbWS.CurrentCamera:WorldToViewportPoint(pos)
    if not on or vp.Z<0 then return false end
    local c=_rbWS.CurrentCamera.ViewportSize/2
    return (Vector2.new(vp.X,vp.Y)-c).Magnitude<=getgenv().ASCfg.FOVRadius
end
local function _asVisible(from,to,model)
    local rp=RaycastParams.new()
    rp.FilterDescendantsInstances={_rbPlayer.Character}
    rp.FilterType=Enum.RaycastFilterType.Exclude
    local res=_rbWS:Raycast(from,(to-from)*0.99,rp)
    if not res then return true end
    return res.Instance:IsDescendantOf(model)
end
local function _asHit(e)
    local hp=getgenv().ASCfg.HitPart
    if hp=="Head" then return e.Head end
    if hp=="Torso" then return e.Torso or e.Head end
    local pool={}
    for _,n in ipairs({"Head","UpperTorso","Torso","LowerTorso","LeftUpperArm","RightUpperArm","LeftUpperLeg","RightUpperLeg"}) do
        local p=e.Model:FindFirstChild(n); if p then table.insert(pool,p) end
    end
    return pool[math.random(1,#pool)] or e.Head
end
local function _asPred(e,p)
    if not getgenv().ASCfg.PredictEnabled then return p.Position end
    return p.Position+e.Root.AssemblyLinearVelocity*getgenv().ASCfg.PredictTime
end
local asLock=nil
local function _asVal()
    if not asLock then return false end
    local h=asLock.Hum; local r=asLock.Root
    if not h or not h.Parent or h.Health<=0 then asLock=nil; return false end
    local ch=_rbPlayer.Character; local my=ch and ch:FindFirstChild("HumanoidRootPart")
    if not my then asLock=nil; return false end
    if (my.Position-r.Position).Magnitude>getgenv().ASCfg.MaxDistance then asLock=nil; return false end
    return true
end
local function _asBest()
    local c=getgenv().ASCfg
    if c.TargetLock and _asVal() then return asLock end
    asLock=nil
    local ch=_rbPlayer.Character; local my=ch and ch:FindFirstChild("HumanoidRootPart")
    if not my then return nil end
    local now=os.clock()
    if now-asCacheT>AS_TTL then _asRebuild(); asCacheT=now end
    local best,bs=nil,math.huge
    local center=_rbWS.CurrentCamera.ViewportSize/2
    for _,e in ipairs(asCache) do
        local h=e.Hum
        if h and h.Parent and h.Health>0 then
            local skip=false
            if e.IsPlayer then
                if not c.TargetPlayers then skip=true end
                if c.TeamCheck and e.Player and e.Player.Team==_rbPlayer.Team then skip=true end
            end
            if not skip then
                local d=(my.Position-e.Root.Position).Magnitude
                if d<=c.MaxDistance and (not c.FOVEnabled or _asInFov(e.Head.Position)) then
                    if not c.VisibleOnly or _asVisible(my.Position,e.Head.Position,e.Model) then
                        local sc
                        if c.FOVEnabled then
                            local vp=_rbWS.CurrentCamera:WorldToViewportPoint(e.Head.Position)
                            sc=(Vector2.new(vp.X,vp.Y)-center).Magnitude
                        else sc=d end
                        if sc<bs then bs=sc; best=e end
                    end
                end
            end
        end
    end
    if best and c.TargetLock then asLock=best end
    return best
end
local function _asFire(e,lf)
    if not _rbUtil or not _rbEnum then return end
    local c=getgenv().ASCfg
    local hp=_asHit(e); if not hp then return end
    local tp=_asPred(e,hp)
    local off=Vector3.new((math.random()-0.5)*c.Spread,(math.random()-0.5)*c.Spread,(math.random()-0.5)*c.Spread)
    local aimed=tp+off
    local aimCF=CFrame.lookAt(aimed+Vector3.new(0,1,0),aimed)
    local objOff=hp.CFrame:ToObjectSpace(CFrame.new(aimed))
    local data={}
    data[utf8.char(1)]={
        [utf8.char(0)]=_rbUtil:EncodeCFrame(aimCF),
        [utf8.char(1)]=_rbUtil:EncodeCFrame(hp.CFrame),
        [utf8.char(2)]=hp,
        [utf8.char(3)]=_rbUtil:EncodeCFrame(objOff),
    }
    pcall(function() _rbRS.Remotes.Replication.Fighter.UseItem:FireServer(lf.EquippedItem:Get("ObjectID"),_rbEnum:ToEnum("StartShooting"),data,nil) end)
end

local function saveConfig()
    local data = {}
    for k, v in pairs(S) do
        if typeof(v) == "Color3" then
            data[k] = {__type = "Color3", r = v.R, g = v.G, b = v.B}
        elseif typeof(v) == "EnumItem" then
            data[k] = {__type = "Enum", enum = tostring(v.EnumType), name = v.Name}
        elseif type(v) ~= "function" and typeof(v) ~= "RBXScriptConnection" and typeof(v) ~= "Instance" and typeof(v) ~= "userdata" then
            data[k] = v
        end
    end
    local json = HttpService:JSONEncode(data)
    if writefile then
        pcall(writefile, configName, json)
        Library:Notify("é…ç½®å·²ä¿å­˜", "success")
    else
        Library:Notify("æ— æ³•ä¿å­˜é…ç½®", "warning")
    end
end

        if not silent then Library:Notify("é…ç½®è§£æžå¤±è´¥", "warning") end
